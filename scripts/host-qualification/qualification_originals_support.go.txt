//go:build qualification

// Qualification support for the optional lyrics-variants capability. It is
// overlaid into the host's plugins package by scripts/host-qualification; it is
// never part of a production build.
package plugins

import (
	"archive/zip"
	"database/sql"
	"encoding/json"
	"errors"
	"net/http"
	"os"
	"path/filepath"
	"testing"
	"time"

	"github.com/navidrome/navidrome/conf"
	"github.com/navidrome/navidrome/model"
)

// QualificationPlugin is a real loaded plugin package with its own KV store and
// task workers, plus test-only expiry helpers.
type QualificationPlugin struct {
	Manager   *Manager
	PluginID  string
	kv        *kvstoreServiceImpl
	tasks     *taskQueueServiceImpl
	transport http.RoundTripper
	catalog   []string
	path      string
}

// LoadQualificationPlugin loads a production .ndp into a real Manager with
// disposable host services and a deterministic HTTP transport. Only the named
// sources are enabled; the manifest supplies the full source catalog.
func LoadQualificationPlugin(t testing.TB, packagePath string, sourceIDs []string, transport http.RoundTripper) *QualificationPlugin {
	t.Helper()
	catalog := manifestSourceIDs(t, packagePath)
	dir := t.TempDir()
	t.Cleanup(conf.SnapshotConfig())
	conf.Update(func(options *conf.Options) {
		options.DataFolder = conf.NewDir(filepath.Join(dir, "data"))
		options.CacheFolder = conf.NewDir(filepath.Join(dir, "cache"))
		options.Plugins.Folder = conf.NewDir(filepath.Join(dir, "packages"))
		options.Plugins.Enabled = true
		options.Plugins.AutoReload = false
		options.Plugins.LogLevel = "error"
		// The qualification host resolves exactly this plugin for lyrics.
		options.LyricsPriority = "nd-better-lyrics"
	})
	manager := &Manager{
		plugins:        make(map[string]*plugin),
		metrics:        noopMetricsRecorder{},
		subsonicRouter: http.NotFoundHandler(),
	}
	if err := manager.initRuntime(t.Context(), filepath.Join(dir, "compiled")); err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() {
		if err := manager.Stop(); err != nil {
			t.Error(err)
		}
	})

	enabled := make(map[string]bool, len(sourceIDs))
	for _, id := range sourceIDs {
		enabled[id] = true
	}
	rows := make([]map[string]any, 0, len(catalog))
	for _, id := range catalog {
		rows = append(rows, map[string]any{"id": id, "enabled": enabled[id]})
	}
	config, err := json.Marshal(map[string]any{"sources": rows})
	if err != nil {
		t.Fatal(err)
	}
	const pluginID = "nd-better-lyrics"
	if err := manager.loadPluginWithConfig(&model.Plugin{
		ID: pluginID, Path: packagePath, Enabled: true, AllUsers: true, Config: string(config),
	}); err != nil {
		t.Fatal(err)
	}
	loaded := manager.plugins[pluginID]
	result := &QualificationPlugin{Manager: manager, PluginID: pluginID, transport: transport, catalog: catalog, path: packagePath}
	result.bind(loaded)
	if result.kv == nil {
		t.Fatal("real plugin KV service was not initialized")
	}
	return result
}

func (h *QualificationPlugin) bind(loaded *plugin) {
	for _, closer := range loaded.closers {
		switch service := closer.(type) {
		case *httpServiceImpl:
			service.client.Transport = h.transport
		case *kvstoreServiceImpl:
			h.kv = service
		case *taskQueueServiceImpl:
			h.tasks = service
		}
	}
}

// Configure reloads the real plugin with a new source/key configuration while
// retaining the same KV database, so generation and retained-state behavior are
// exercised through the production reload path.
func (h *QualificationPlugin) Configure(t testing.TB, sources []string, apiKey string) {
	t.Helper()
	enabled := make(map[string]bool, len(sources))
	for _, id := range sources {
		enabled[id] = true
	}
	rows := make([]map[string]any, 0, len(h.catalog))
	for _, id := range h.catalog {
		rows = append(rows, map[string]any{"id": id, "enabled": enabled[id]})
	}
	config := map[string]any{"sources": rows}
	if apiKey != "" {
		config["betterLyricsApiKey"] = apiKey
	}
	raw, err := json.Marshal(config)
	if err != nil {
		t.Fatal(err)
	}
	if err := h.Manager.unloadPlugin(h.PluginID); err != nil {
		t.Fatal(err)
	}
	if err := h.Manager.loadPluginWithConfig(&model.Plugin{
		ID: h.PluginID, Path: h.path, Enabled: true, AllUsers: true, Config: string(raw),
	}); err != nil {
		t.Fatal(err)
	}
	loaded := h.Manager.plugins[h.PluginID]
	h.bind(loaded)
	if h.kv == nil {
		t.Fatal("real plugin KV service was not initialized after reload")
	}
}

// Idle waits for real background tasks to settle, bounded.
func (h *QualificationPlugin) Idle(t testing.TB) {
	t.Helper()
	deadline := time.Now().Add(5 * time.Second)
	for {
		var pending int
		if err := h.tasks.db.QueryRowContext(t.Context(), `SELECT COUNT(*) FROM tasks WHERE status IN ('pending','running')`).Scan(&pending); err != nil {
			t.Fatal(err)
		}
		if pending == 0 {
			return
		}
		if time.Now().After(deadline) {
			t.Fatalf("task queue still has %d pending/running tasks", pending)
		}
		time.Sleep(20 * time.Millisecond)
	}
}

// Keys lists test-owned KV keys under a prefix.
func (h *QualificationPlugin) Keys(t testing.TB, prefix string) []string {
	t.Helper()
	keys, err := h.kv.List(t.Context(), prefix)
	if err != nil {
		t.Fatal(err)
	}
	return keys
}

// KVPath is the real plugin KV database path, for retained-state compatibility
// tests.
func (h *QualificationPlugin) KVPath() string {
	return filepath.Join(conf.Server().DataFolder.String(), "plugins", h.PluginID, "kvstore.db")
}

// ExportKV checkpoints and copies the real plugin KV database.
func (h *QualificationPlugin) ExportKV(t testing.TB, destination string) {
	t.Helper()
	if _, err := h.kv.db.ExecContext(t.Context(), `PRAGMA wal_checkpoint(TRUNCATE)`); err != nil {
		t.Fatal(err)
	}
	data, err := os.ReadFile(h.KVPath())
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(destination, data, 0600); err != nil {
		t.Fatal(err)
	}
}

// PublicationTrackKey exposes the host's admission track identity.
func (h *QualificationPlugin) PublicationTrackKey(mf *model.MediaFile) string {
	return publicationTrackKey(mf)
}

// Admission returns the current admission identity for a track.
func (h *QualificationPlugin) Admission(t testing.TB, trackKey string) (string, string, bool) {
	t.Helper()
	var generation, token string
	err := h.kv.db.QueryRowContext(t.Context(), `SELECT generation, token FROM plugin_publication_admissions WHERE track_key=?`, trackKey).Scan(&generation, &token)
	if errors.Is(err, sql.ErrNoRows) {
		return "", "", false
	}
	if err != nil {
		t.Fatal(err)
	}
	return generation, token, true
}

// EnqueueTask injects a deterministic task payload into the real queue.
func (h *QualificationPlugin) EnqueueTask(t testing.TB, queue string, payload []byte) string {
	t.Helper()
	id, err := h.tasks.Enqueue(t.Context(), queue, payload)
	if err != nil {
		t.Fatal(err)
	}
	return id
}

// Delete removes test-owned KV keys under a prefix.
func (h *QualificationPlugin) Delete(t testing.TB, prefix string) {
	t.Helper()
	if _, err := h.kv.DeleteByPrefix(t.Context(), prefix); err != nil {
		t.Fatal(err)
	}
}

func manifestSourceIDs(t testing.TB, packagePath string) []string {
	t.Helper()
	archive, err := zip.OpenReader(packagePath)
	if err != nil {
		t.Fatal(err)
	}
	defer archive.Close()
	var manifest struct {
		Config struct {
			Schema struct {
				Properties struct {
					Sources struct {
						Default []struct {
							ID string `json:"id"`
						} `json:"default"`
					} `json:"sources"`
				} `json:"properties"`
			} `json:"schema"`
		} `json:"config"`
	}
	for _, entry := range archive.File {
		if entry.Name != "manifest.json" {
			continue
		}
		reader, err := entry.Open()
		if err != nil {
			t.Fatal(err)
		}
		err = json.NewDecoder(reader).Decode(&manifest)
		reader.Close()
		if err != nil {
			t.Fatal(err)
		}
	}
	ids := make([]string, 0, len(manifest.Config.Schema.Properties.Sources.Default))
	for _, row := range manifest.Config.Schema.Properties.Sources.Default {
		ids = append(ids, row.ID)
	}
	if len(ids) == 0 {
		t.Fatal("plugin manifest has no source defaults")
	}
	return ids
}
